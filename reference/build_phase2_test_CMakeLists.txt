# Aggregate module implementations are compiled into distinct correctness and
# benchmark executables.  Stage-6 CTest registration keeps their acceptance
# roles separate and gates every benchmark behind its correctness executable.
function(add_phase2_correctness_variant correctness_target benchmark_target source_file)
  add_executable(${correctness_target} ${source_file})
  get_target_property(benchmark_link_libraries ${benchmark_target} LINK_LIBRARIES)
  target_link_libraries(${correctness_target} ${benchmark_link_libraries})
  target_compile_definitions(${correctness_target} PRIVATE __HIP_PLATFORM_AMD__ HIP_CORRECTNESS_MODE=1)
endfunction()
